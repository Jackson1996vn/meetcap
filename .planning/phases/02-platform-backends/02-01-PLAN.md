   ```python
   """Spike: Validate ScreenCaptureKit + PyObjC CMSampleBuffer extraction on macOS 26.x."""
   import time
   import threading
   import numpy as np
   import objc
   from objc import super
   from Foundation import NSObject, NSRunLoop, NSDate
   from ScreenCaptureKit import (
       SCStream, SCShareableContent, SCStreamConfiguration, SCContentFilter,
       SCStreamOutputTypeAudio, SCStreamOutputTypeScreen,
   )
   from CoreMedia import CMSampleBufferGetNumSamples, CMSampleBufferGetFormatDescription
   from AVFoundation import CMSampleBufferGetAudioBufferListWithRetainedBlockBuffer

   received = []

   class SpikeDelegate(NSObject):
       def initWithEvent_(self, event):
           self = super().init()
           if self is None:
               return None
           self._event = event
           return self

       def stream_didOutputSampleBuffer_ofType_(self, stream, sampleBuffer, output_type):
           if output_type == SCStreamOutputTypeScreen:
               return
           if output_type == SCStreamOutputTypeAudio:
               n_samples = CMSampleBufferGetNumSamples(sampleBuffer)
               fmt = CMSampleBufferGetFormatDescription(sampleBuffer)
               print(f"  Audio callback: {n_samples} samples, fmt={fmt}")
               if fmt is not None:
                   asbd = fmt.audioStreamBasicDescription()
                   print(f"  ASBD: rate={asbd.mSampleRate}, ch={asbd.mChannelsPerFrame}, "
                         f"bitsPerCh={asbd.mBitsPerChannel}, fmt={asbd.mFormatFlags}")
               # Try CMSampleBufferGetAudioBufferListWithRetainedBlockBuffer
               try:
                   result = CMSampleBufferGetAudioBufferListWithRetainedBlockBuffer(
                       sampleBuffer, None, None, 0, None, None, 0, None
                   )
                   print(f"  AudioBufferList result type: {type(result)}")
                   print(f"  AudioBufferList result repr: {repr(result)[:500]}")
                   # Attempt to extract numpy array from result
                   # Document whatever works here
               except Exception as e:
                   print(f"  AudioBufferList extraction failed: {e}")
               received.append(n_samples)

       def stream_didStopWithError_(self, stream, error):
           print(f"  Stream stopped: {error}")

   def run_spike():
       ready = threading.Event()
       content_holder = [None]

       def on_content(content, error):
           if error:
               print(f"ERROR getting content: {error}")
               ready.set()
               return
           displays = content.displays()
           print(f"Displays found: {len(displays)}")
           if not displays:
               print("NO DISPLAYS — Screen Recording permission missing")
               print("Attempting second call to trigger TCC dialog (per D-09)...")
               # Second call triggers TCC dialog
               SCShareableContent.getShareableContentWithCompletionHandler_(on_content_retry)
               return
           content_holder[0] = content
           ready.set()

       def on_content_retry(content, error):
           if error:
               print(f"Retry ERROR: {error}")
           else:
               displays = content.displays()
               print(f"Retry displays: {len(displays)}")
               if displays:
                   content_holder[0] = content
           ready.set()

       SCShareableContent.getShareableContentWithCompletionHandler_(on_content)
       ready.wait(timeout=30)

       content = content_holder[0]
       if content is None:
           print("SPIKE FAILED: No content available")
           return

       display = content.displays()[0]
       config = SCStreamConfiguration.alloc().init()
       config.setCapturesAudio_(True)
       config.setExcludesCurrentProcessAudio_(False)
       config.setSampleRate_(48000)
       config.setChannelCount_(2)

       content_filter = SCContentFilter.alloc().initWithDisplay_excludingApplications_exceptingWindows_(
           display, [], []
       )

       evt = threading.Event()
       delegate = SpikeDelegate.alloc().initWithEvent_(evt)

       stream = SCStream.alloc().initWithFilter_configuration_delegate_(
           content_filter, config, delegate
       )

       stream.addStreamOutput_type_sampleHandlerQueue_error_(
           delegate, SCStreamOutputTypeScreen, None, None
       )
       stream.addStreamOutput_type_sampleHandlerQueue_error_(
           delegate, SCStreamOutputTypeAudio, None, None
       )

       start_ok = threading.Event()
       def on_start(error):
           if error:
               print(f"Start error: {error}")
           else:
               print("Capture started OK")
           start_ok.set()

       stream.startCaptureWithCompletionHandler_(on_start)

       # Spin run loop on background thread
       running = [True]
       def run_loop():
           while running[0]:
               NSRunLoop.currentRunLoop().runUntilDate_(
                   NSDate.dateWithTimeIntervalSinceNow_(0.1)
               )
       t = threading.Thread(target=run_loop, daemon=True)
       t.start()

       start_ok.wait(timeout=10)
       print("Waiting 5 seconds for audio callbacks...")
       time.sleep(5)

       running[0] = False
       stream.stopCaptureWithCompletionHandler_(lambda e: None)
       t.join(timeout=5)

       print(f"\nSPIKE RESULTS:")
       print(f"  Callbacks received: {len(received)}")
       if received:
           print(f"  Sample counts: {received[:5]}...")
           print("  SPIKE PASSED: CMSampleBuffer extraction works on this macOS version")
       else:
           print("  SPIKE FAILED: No audio callbacks received")
           print("  Check: Screen Recording permission, audio playing, run loop")

   if __name__ == "__main__":
       run_spike()
   ```

**CaptureDelegate class (NSObject subclass):**
- `initWithCallback_(self, callback)` -- store callback as `self._callback`, return self. Uses `super().init()` (from objc import super).
- `stream_didOutputSampleBuffer_ofType_(self, stream, sampleBuffer, output_type)`:
  - If `output_type == SCStreamOutputTypeScreen`: return immediately (discard screen frames, required to avoid -3805 per Pitfall 2)
  - If `output_type == SCStreamOutputTypeAudio`: extract audio data from CMSampleBuffer, convert to numpy float32, mix to mono if stereo, call `self._callback(arr)`
  - DO NOT decorate with `@objc.python_method` -- this is called by Objective-C (Pitfall 1, the #647 root cause)
- `stream_didStopWithError_(self, stream, error)`:
  - Log warning with error details (per D-10: do not propagate, do not kill recording)
  - DO NOT decorate with `@objc.python_method`

**CMSampleBuffer to numpy extraction (in CaptureDelegate or helper):**
- Use the extraction pattern validated in the Task 0 spike
- Use `CMSampleBufferGetAudioBufferListWithRetainedBlockBuffer` to extract audio bytes
- Get format description via `CMSampleBufferGetFormatDescription(sampleBuffer)` to read channel count and sample format
- Convert raw bytes to numpy float32 array
- If stereo (2 channels): reshape to (-1, 2) and mean across axis=1 to produce mono (per D-12)
- Apply the exact PyObjC bridge pattern discovered in the spike (Task 0 results document the working approach)

**MacOSAudioBackend class:**
- `__init__(self)`: initialize `self._stream = None`, `self._delegate = None`, `self._thread = None`, `self._running = False`
- `start(self, callback: AudioCallback) -> None`:
  1. Call `SCShareableContent.getShareableContentWithCompletionHandler_` and wait for the completion handler (use `threading.Event` to synchronize)
  2. In the completion handler: check if `content.displays()` is non-empty. If empty, this means Screen Recording permission is likely missing (Pitfall 3).
  3. **D-09 dialog-trigger-and-retry behavior:** If displays list is empty on the first call:
     a. Log info: "No displays found — attempting to trigger Screen Recording permission dialog..."
     b. Call `SCShareableContent.getShareableContentWithCompletionHandler_` a SECOND time. This second invocation triggers the macOS TCC permission dialog (per RESEARCH.md Pitfall 3 and D-09).
     c. Wait up to 30 seconds for the user to approve (use `threading.Event.wait(timeout=30)`).
     d. Check the second call's `content.displays()`. If STILL empty after retry, THEN raise `PermissionError` with message: "Screen Recording permission required. Open System Settings > Privacy & Security > Screen Recording and enable your terminal app, then restart."
     e. If the retry succeeds (displays non-empty), proceed normally with the retry's content.
  4. If displays are available (first call or after successful retry): create SCStreamConfiguration:
     - `config.setCapturesAudio_(True)`
     - `config.setExcludesCurrentProcessAudio_(False)`
     - `config.setSampleRate_(48000)` (per D-11 -- ScreenCaptureKit resamples OS-side)
     - `config.setChannelCount_(2)` (capture stereo, mix to mono in callback)
     - Set `config.setMinimumFrameInterval_` to CMTime(1, 1) (1 fps) to minimize screen capture overhead
  5. Create SCContentFilter for full display: `SCContentFilter.alloc().initWithDisplay_excludingApplications_exceptingWindows_(display, [], [])`
  6. Create delegate: `self._delegate = CaptureDelegate.alloc().initWithCallback_(callback)` -- stored as instance attribute (D-07 retain fix)
  7. Create stream: `self._stream = SCStream.alloc().initWithFilter_configuration_delegate_(content_filter, config, self._delegate)`
  8. Add BOTH outputs (D-08, Pitfall 2):
     - `self._stream.addStreamOutput_type_sampleHandlerQueue_error_(self._delegate, SCStreamOutputTypeScreen, None, None)`
     - `self._stream.addStreamOutput_type_sampleHandlerQueue_error_(self._delegate, SCStreamOutputTypeAudio, None, None)`
  9. Start capture: `self._stream.startCaptureWithCompletionHandler_(self._on_start_complete)`
  10. Start a background thread that spins `NSRunLoop.currentRunLoop().run()` for callback delivery (D-08, Pitfall 5). The thread function should call `NSRunLoop.currentRunLoop().runUntilDate_(NSDate.distantFuture())` in a while loop checking `self._running`.
  11. Set `self._running = True`

- `stop(self) -> None`:
  1. Set `self._running = False`
  2. Call `self._stream.stopCaptureWithCompletionHandler_(lambda error: None)` if stream exists
  3. Join the background thread if it exists
  4. Set `self._stream = None`, `self._delegate = None`

- `_on_start_complete(self, error)`: if error, log warning (D-10). Otherwise log info "System audio capture started".

**Logging:** Use `logging.getLogger(__name__)` throughout. Log at INFO for start/stop, WARNING for errors (per D-10).

**CRITICAL anti-patterns to avoid (from RESEARCH.md):**
- Do NOT use `@objc.python_method` on delegate methods (Pitfall 1)
- Do NOT skip the screen output registration (Pitfall 2 -- causes -3805)
- Do NOT use `objc.retain()` or `objc.release()` -- they do not exist in modern PyObjC
- Do NOT store delegate in a local variable -- must be instance attribute (D-07)
- Do NOT immediately raise PermissionError on empty displays -- must attempt TCC dialog trigger and retry first (D-09)
